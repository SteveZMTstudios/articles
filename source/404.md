    function searchGoogle() {

            window.open(`https://www.google.com/search?q=${encodeURIComponent(query)}`, '_blank');

<script data-pjax>
    // 404 URL 变动自动猜测与回落系统
    (function() {
        const path = window.location.pathname;
        const segments = path.split('/').filter(Boolean);
        if (segments.length > 0) {
            const slug = segments[segments.length - 1]; // 提取路径最后一段标识符
            // 忽略带有常见静态扩展名的情况，以免对资源 404 进行误判跳转
            if (slug.includes('.')) return; 

            fetch('/search.xml')
                .then(res => res.text())
                .then(str => new window.DOMParser().parseFromString(str, "text/xml"))
                .then(data => {
                    const entries = data.getElementsByTagName("entry");
                    for (let i = 0; i < entries.length; i++) {
                        const urlNode = entries[i].getElementsByTagName("url")[0];
                        if (!urlNode) continue;
                        const url = urlNode.textContent;
                        
                        // 匹配 slug（确保是一整段）
                        if (url.endsWith(`/${slug}/`) || url.endsWith(`/${slug}.html`) || url.includes(`/${slug}/`)) {
                            const titleNode = entries[i].getElementsByTagName("title")[0];
                            const title = titleNode ? titleNode.textContent : "目标页面";
                            
                            const container = document.getElementById("auto-redirect-container");
                            if (container && !container.innerHTML) {
                                container.innerHTML = `
                                    <div class="mdui-card mdui-shadow-2 mdui-ripple">
                                        <div class="mdui-card-primary">
                                            <div class="mdui-card-primary-title">发现可能的页面变动</div>
                                            <div class="mdui-card-primary-subtitle">这是否是您要找的页面？</div>
                                        </div>
                                        <div class="mdui-card-content">
                                            <span style="font-size: 1.5em;">${title}</span><br>
                                            <a href="${url}" class="mdui-text-color-theme-secondary" style="word-break: break-all;">${url}</a>
                                            <hr>
                                            <a href="${url}" class="mdui-btn mdui-ripple mdui-color-theme-accent">立即前往</a>
                                        
                                    </div>
                                `;
                                if (window.mdui && window.mdui.mutation) {
                                    window.mdui.mutation();
                                }
                            }
                            
                            break;
                        }
                    }
                })
                .catch(e => console.log("加载搜索索引进行 404 恢复失败", e));
        }
    })();
</script>
